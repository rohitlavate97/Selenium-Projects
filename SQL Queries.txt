);

    order_id INT PRIMARY KEY,
    customer_id INT,
    product VARCHAR(50),
    amount DECIMAL(10,2)

-- inner join
SELECT c.customer_name, o.product, o.amount
FROM customers c
INNER JOIN orders o
ON c.customer_id = o.customer_id;

-- left join
SELECT c.customer_name, o.product
FROM customers c
LEFT JOIN orders o
ON c.customer_id = o.customer_id;

-- right join
SELECT c.customer_name, o.product
FROM customers c
RIGHT JOIN orders o
ON c.customer_id = o.customer_id;

-- full join
SELECT c.customer_name, o.product
FROM customers c
LEFT JOIN orders o ON c.customer_id = o.customer_id

UNION

SELECT c.customer_name, o.product
FROM customers c
RIGHT JOIN orders o ON c.customer_id = o.customer_id;

-- cross join
SELECT c.customer_name, o.product
FROM customers c
CROSS JOIN orders o;

SQL Constraints:
following constraints:
• NOT NULL
• UNIQUE : uniquely identifies each record in a database table,UNIQUE and PRIMARY KEY constraints both provide a guarantee for uniqueness for a column or 
set of columns.
CREATE TABLE Persons (
    P_Id INT NOT NULL,
    LastName VARCHAR(255) NOT NULL,
    FirstName VARCHAR(255),
    Address VARCHAR(255),
    City VARCHAR(255),
    UNIQUE (P_Id)
);
• PRIMARY KEY
CREATE TABLE Persons (
    P_Id INT NOT NULL,
    LastName VARCHAR(255) NOT NULL,
    FirstName VARCHAR(255),
    Address VARCHAR(255),
    City VARCHAR(255),
    PRIMARY KEY (P_Id)
);

• FOREIGN KEY
CREATE TABLE Orders (
    O_Id INT NOT NULL,
    OrderNo INT NOT NULL,
    P_Id INT,
    PRIMARY KEY (O_Id),
    FOREIGN KEY (P_Id) REFERENCES Persons(P_Id)
);

• CHECK: CHECK constraint is used to limit the value range that can be placed in a column.
CREATE TABLE Persons (
    P_Id INT NOT NULL,
    LastName VARCHAR(255) NOT NULL,
    FirstName VARCHAR(255),
    Address VARCHAR(255),
    City VARCHAR(255),
    CHECK (P_Id > 0)
);

• DEFAULT: used to insert a default value into a column
CREATE TABLE Persons (
    P_Id INT NOT NULL,
    LastName VARCHAR(255) NOT NULL,
    FirstName VARCHAR(255),
    Address VARCHAR(255),
    City VARCHAR(255) DEFAULT 'Sandnes'
);
